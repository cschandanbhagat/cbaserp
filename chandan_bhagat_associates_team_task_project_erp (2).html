Automatically sync your ERP tasks, clients register, and payment recovery logs into dedicated tabs in your Google Sheet.

Tab 1: ERP_Tasks

Alternative Spreadsheet Import / Export (No Coding Required)

Google Apps Script Code (Paste in Apps Script Editor & Deploy as Web App): 
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Tasks Tab
  var sheetTasks = ss.getSheetByName("ERP_Tasks") || ss.insertSheet("ERP_Tasks");
  sheetTasks.clear();
  sheetTasks.appendRow(["Task ID", "Client Name", "Title", "Vertical", "Assignee", "Branch", "Priority", "Status", "Due Date", "Completed Date", "Notes"]);
  data.tasks.forEach(function(t) {
    sheetTasks.appendRow([t.id, t.clientName, t.title, t.project, t.assignee, t.branch, t.priority, t.status, t.dueDate, t.completedDate, t.notes]);
  });
  
  // 2. Client Register Tab
  var sheetClients = ss.getSheetByName("Client_Register") || ss.insertSheet("Client_Register");
  sheetClients.clear();
  sheetClients.appendRow(["S.No.", "Client ID", "Entity Name", "Type", "CIN", "PAN", "GSTIN", "Signatory", "Mobile", "Email", "Branch", "Services", "Status", "Notes"]);
  data.clients.forEach(function(c) {
    sheetClients.appendRow([c.sno, c.id, c.name, c.type, c.cin, c.pan, c.gstin, c.contactPerson, c.phone, c.email, c.branch, c.services, c.status, c.notes]);
  });
  
  // 3. Payment Recovery Tab
  var sheetRec = ss.getSheetByName("Payment_Recovery") || ss.insertSheet("Payment_Recovery");
  sheetRec.clear();
  sheetRec.appendRow(["S.No.", "Client Name", "Phone", "Billed (₹)", "Received (₹)", "Balance Due (₹)", "Promise Date", "Field Visit Required", "Missed Calls", "Last Call Log"]);
  data.recovery.forEach(function(r) {
    sheetRec.appendRow([r.sno, r.clientName, r.phone, r.billed, r.received, r.due, r.promiseDate, r.fieldVisit, r.missedCalls, r.lastLog]);
  });
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
